find_package(Qt6 REQUIRED COMPONENTS Test)

function(pou_add_test name)
    qt_add_executable(${name} ${name}.cpp ${ARGN})
    target_link_libraries(${name} PRIVATE pou_core Qt6::Test)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
