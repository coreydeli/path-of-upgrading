find_package(Qt6 REQUIRED COMPONENTS Test QuickTest)

function(pou_add_test name)
    qt_add_executable(${name} ${name}.cpp ${ARGN})
    target_link_libraries(${name} PRIVATE pou_core Qt6::Test)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

function(pou_add_services_test name)
    pou_add_test(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE pou_services)
endfunction()

function(pou_add_platform_test name)
    pou_add_test(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE pou_platform)
endfunction()

function(pou_add_app_test name)
    pou_add_test(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE pou_app)
endfunction()

# for tests that load the app's QML, offscreen
function(pou_add_qml_test name)
    pou_add_app_test(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE pou_appplugin Qt6::QuickTest)
    set_target_properties(${name} PROPERTIES QT_QML_MODULE_NO_IMPORT_SCAN ON)
endfunction()

# for tests with a stand-in for Path of Building, see fakepob.h
function(pou_use_fakepob name)
    target_sources(${name} PRIVATE fakepob.h)
    target_compile_definitions(${name} PRIVATE FAKEPOB="$<TARGET_FILE:fakepob>")
    add_dependencies(${name} fakepob)
endfunction()
